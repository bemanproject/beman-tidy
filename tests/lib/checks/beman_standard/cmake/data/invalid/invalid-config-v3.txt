cmake_minimum_required(VERSION 3.30...4.3)

project(
    beman.exemplar
    DESCRIPTION "A Beman Library Exemplar"
    LANGUAGES CXX
    VERSION 0.1.0
)

include(infra/cmake/beman-install-library.cmake)

add_library(beman.exemplar INTERFACE)

add_subdirectory(include/beman/exemplar)

# beman_install_library() called for a different library whose name has the library name as a prefix.
beman_install_library(beman.exemplar-extra TARGETS beman.exemplar)

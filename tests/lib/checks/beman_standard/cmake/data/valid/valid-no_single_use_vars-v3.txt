cmake_minimum_required(VERSION 3.30...4.3)

project(
    beman.exemplar
    DESCRIPTION "A Beman Library Exemplar"
    LANGUAGES CXX
    VERSION 0.1.0
)

add_library(beman.exemplar INTERFACE)
add_library(beman::exemplar ALIAS beman.exemplar)

# A variable assigned on several branches is not a single-use constant.
if(WIN32)
    set(PLATFORM_DIR windows)
else()
    set(PLATFORM_DIR posix)
endif()
target_include_directories(beman.exemplar INTERFACE ${PLATFORM_DIR})

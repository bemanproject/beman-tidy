cmake_minimum_required(VERSION 3.30...4.3)

project(
    beman.exemplar
    DESCRIPTION "A Beman Library Exemplar"
    LANGUAGES CXX
    VERSION 0.1.0
)

add_library(beman.exemplar INTERFACE)
add_library(beman::exemplar ALIAS beman.exemplar)

# CMake built-in variables are read by CMake itself, so one reference is fine.
if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.28)
    set(CMAKE_CXX_SCAN_FOR_MODULES ON)
endif()
message(STATUS "modules scan: ${CMAKE_CXX_SCAN_FOR_MODULES}")
set(BUILD_SHARED_LIBS OFF)
message(STATUS "shared: ${BUILD_SHARED_LIBS}")
